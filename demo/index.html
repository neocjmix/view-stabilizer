03 / API

One element. Six methods.

import { createViewStabilizer } from "@neocjmix/view-stabilizer";

const stabilizer = createViewStabilizer(element, {
  viewingDistance: 400,
  maxTilt: 65,
  smoothing: 0.12,
});

await stabilizer.requestPermission(); // call in a user gesture on iOS
stabilizer.start();
stabilizer.calibrate();               // canonical API
stabilizer.recenter();                // readable alias
